Nota: El boton azul flotante en la esquina inferior derecha es el widget. Haz clic en el para abrir el chat.

Codigo de Integracion

// Cargar el widget
import { ChatbotWidget } from 'chatbot-widget';

// O via script tag
// <script src="chatbot-widget.umd.js"></script>

// Inicializar
ChatbotWidget.init({
  projectId: 'mi-proyecto',
  endpoints: {
    websocket: 'wss://api.ejemplo.com/chat',
  },
  bot: {
    name: 'Soporte',
  },
});
